cmake_minimum_required(VERSION 3.16)
project(SIMPL_R_Liquid_v02 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_library(liquid_engine src/LiquidEngine.cpp)
target_include_directories(liquid_engine PUBLIC include)
add_library(liquid_wav src/Wav.cpp src/AudioImport.cpp)
target_include_directories(liquid_wav PUBLIC include src)
target_link_libraries(liquid_wav PUBLIC liquid_engine)
add_executable(liquid_mac src/main.cpp)
target_include_directories(liquid_mac PRIVATE src)
target_link_libraries(liquid_mac PRIVATE liquid_wav)
if(APPLE)
  target_link_libraries(liquid_wav PRIVATE "-framework AudioToolbox" "-framework CoreFoundation")
  target_link_libraries(liquid_mac PRIVATE "-framework AudioToolbox" "-framework CoreFoundation")
